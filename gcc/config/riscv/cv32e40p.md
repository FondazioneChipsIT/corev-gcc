(define_automaton "cv32e40p_pipe")

(define_cpu_unit "cv32e40p_alu"     "cv32e40p_pipe")
(define_cpu_unit "cv32e40p_imuldiv" "cv32e40p_pipe")
(define_cpu_unit "cv32e40p_fdivsqrt" "cv32e40p_pipe")

;; ---------------------------------------------------------------------
;; Integer ALU.
;; ---------------------------------------------------------------------
(define_insn_reservation "cv32e40p_alu_insn" 1
  (and (eq_attr "tune" "cv32e40p")
       (eq_attr "type" "unknown,const,arith,shift,slt,multi,auipc,nop,logical,\
                         move,bitmanip,min,max,minu,maxu,clz,ctz,rotate,atomic,\
                         condmove,crypto,mvpair,zicond"))
  "cv32e40p_alu")

;; ---------------------------------------------------------------------
;; Load.
;; ---------------------------------------------------------------------
(define_insn_reservation "cv32e40p_load" 2
  (and (eq_attr "tune" "cv32e40p")
       (eq_attr "type" "load,fpload"))
  "cv32e40p_alu")

;; ---------------------------------------------------------------------
;; Store.
;; ---------------------------------------------------------------------
(define_insn_reservation "cv32e40p_store" 1
  (and (eq_attr "tune" "cv32e40p")
       (eq_attr "type" "store,fpstore"))
  "cv32e40p_alu")

;; ---------------------------------------------------------------------
;; FPU move/convert/compare transfers.
;; ---------------------------------------------------------------------
(define_insn_reservation "cv32e40p_xfer" 3
  (and (eq_attr "tune" "cv32e40p")
       (eq_attr "type" "mfc,mtc,fcvt,fcvt_i2f,fcvt_f2i,fmove,fcmp"))
  "cv32e40p_alu")

;; ---------------------------------------------------------------------
;; Branch: modeled at the taken-branch cost (3 cycles).
;; ---------------------------------------------------------------------
(define_insn_reservation "cv32e40p_branch" 3
  (and (eq_attr "tune" "cv32e40p")
       (eq_attr "type" "branch"))
  "cv32e40p_alu")

;; Jump / call / jalr / ret: 2 cycles (taken from ID stage).
(define_insn_reservation "cv32e40p_jump" 2
  (and (eq_attr "tune" "cv32e40p")
       (eq_attr "type" "jump,call,jalr,ret"))
  "cv32e40p_alu")

(define_insn_reservation "cv32e40p_trap" 1
  (and (eq_attr "tune" "cv32e40p")
       (eq_attr "type" "trap"))
  "cv32e40p_alu")

(define_insn_reservation "cv32e40p_sfb_alu" 2
  (and (eq_attr "tune" "cv32e40p")
       (eq_attr "type" "sfb_alu"))
  "cv32e40p_alu")

;; ---------------------------------------------------------------------
;; Multiply.  Hardware distinguishes mul (1 cycle) from
;; mulh/mulhsu/mulhu (5 cycles), but both share the "imul" type in the
;; generic attribute set, so a single reservation can't split them
;; without insn-specific checks.
;; ---------------------------------------------------------------------
(define_insn_reservation "cv32e40p_mulh" 5
  (and (eq_attr "tune" "cv32e40p")
       (match_test "recog_memoized (insn) == CODE_FOR_smulsi3_highpart
                 || recog_memoized (insn) == CODE_FOR_umulsi3_highpart
                 || recog_memoized (insn) == CODE_FOR_usmulsi3_highpart
                 || recog_memoized (insn) == CODE_FOR_smuldi3_highpart
                 || recog_memoized (insn) == CODE_FOR_umuldi3_highpart
                 || recog_memoized (insn) == CODE_FOR_usmuldi3_highpart"))
  "cv32e40p_imuldiv*5")

(define_insn_reservation "cv32e40p_imul" 1
  (and (eq_attr "tune" "cv32e40p")
       (and (eq_attr "type" "imul,clmul,cpop")
            (match_test "recog_memoized (insn) != CODE_FOR_smulsi3_highpart
                      && recog_memoized (insn) != CODE_FOR_umulsi3_highpart
                      && recog_memoized (insn) != CODE_FOR_usmulsi3_highpart
                      && recog_memoized (insn) != CODE_FOR_smuldi3_highpart
                      && recog_memoized (insn) != CODE_FOR_umuldi3_highpart
                      && recog_memoized (insn) != CODE_FOR_usmuldi3_highpart")))
  "cv32e40p_imuldiv")


;; Divide / remainder.
(define_insn_reservation "cv32e40p_idivsi" 35
  (and (eq_attr "tune" "cv32e40p")
       (and (eq_attr "type" "idiv")
            (eq_attr "mode" "SI")))
  "cv32e40p_imuldiv*35")

;; ---------------------------------------------------------------------
;; FPU arithmetic.
;; ---------------------------------------------------------------------
(define_insn_reservation "cv32e40p_fmul_half" 2
  (and (eq_attr "tune" "cv32e40p")
       (and (eq_attr "type" "fadd,fmul,fmadd")
            (eq_attr "mode" "HF")))
  "cv32e40p_alu")

(define_insn_reservation "cv32e40p_fmul_single" 2
  (and (eq_attr "tune" "cv32e40p")
       (and (eq_attr "type" "fadd,fmul,fmadd")
            (eq_attr "mode" "SF")))
  "cv32e40p_alu")

(define_insn_reservation "cv32e40p_fdiv" 19
  (and (eq_attr "tune" "cv32e40p")
       (eq_attr "type" "fdiv"))
  "cv32e40p_fdivsqrt*19")

(define_insn_reservation "cv32e40p_fsqrt" 19
  (and (eq_attr "tune" "cv32e40p")
       (eq_attr "type" "fsqrt"))
  "cv32e40p_fdivsqrt*19")

;; ---------------------------------------------------------------------
;; Hazard bypasses.
;; ---------------------------------------------------------------------

;; Load-use hazard: a load followed immediately by a dependent ALU/branch
;; instruction costs 1 extra cycle vs. the reservation latency above.
(define_bypass 3 "cv32e40p_load" "cv32e40p_alu_insn,cv32e40p_branch,cv32e40p_sfb_alu")

;; jalr-use hazard: jalr depending on the immediately preceding insn's
;; result costs 1 extra cycle.
(define_bypass 2 "cv32e40p_alu_insn" "cv32e40p_jump")
